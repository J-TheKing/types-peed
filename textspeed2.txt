// --- Audio Synthesizer ---
class SoundEffects {
  constructor() {
    this.ctx = null;
    this.enabled = true;
  }

  init() {
    if (!this.ctx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume();
    }
  }

  playKeyClick() {
    if (!this.enabled || !this.ctx) return;
    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    const filter = this.ctx.createBiquadFilter();

    osc.type = "triangle";
    osc.frequency.setValueAtTime(450 + Math.random() * 80, now);
    osc.frequency.exponentialRampToValueAtTime(120, now + 0.04);

    filter.type = "bandpass";
    filter.frequency.setValueAtTime(1400, now);
    filter.Q.setValueAtTime(3, now);

    gain.gain.setValueAtTime(0.3, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.04);

    osc.connect(filter);
    filter.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.045);
  }

  playSpaceClick() {
    if (!this.enabled || !this.ctx) return;
    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = "sine";
    osc.frequency.setValueAtTime(220, now);
    osc.frequency.exponentialRampToValueAtTime(60, now + 0.07);

    gain.gain.setValueAtTime(0.4, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.07);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.08);
  }

  playError() {
    if (!this.enabled || !this.ctx) return;
    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = "sawtooth";
    osc.frequency.setValueAtTime(140, now);
    osc.frequency.linearRampToValueAtTime(95, now + 0.09);

    gain.gain.setValueAtTime(0.25, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.1);
  }

  playFinishChime() {
    if (!this.enabled || !this.ctx) return;
    const notes = [523.25, 659.25, 783.99, 1046.50];
    notes.forEach((freq, idx) => {
      const now = this.ctx.currentTime + idx * 0.09;
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();

      osc.type = "triangle";
      osc.frequency.setValueAtTime(freq, now);

      gain.gain.setValueAtTime(0.28, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.28);

      osc.connect(gain);
      gain.connect(this.ctx.destination);

      osc.start(now);
      osc.stop(now + 0.3);
    });
  }
}

function updateCaretPosition() {
  const currentWordEl = wordsContainer.children[state.currentWordIdx];
  if (!currentWordEl) return;

  const letters = currentWordEl.querySelectorAll(".letter");
  let targetLeft = 0;
  let targetTop = 0;

  if (state.currentLetterIdx < letters.length) {
    const letter = letters[state.currentLetterIdx];
    targetLeft = letter.offsetLeft;
    targetTop = letter.offsetTop;
  } else {
    const lastLetter = letters[letters.length - 1];
    if (lastLetter) {
      targetLeft = lastLetter.offsetLeft + lastLetter.offsetWidth;
      targetTop = lastLetter.offsetTop;
    }
  }

  caretEl.style.left = `${targetLeft}px`;
  caretEl.style.top = `${targetTop}px`;

  const lineHeight = 42;
  const scrollOffset = targetTop - lineHeight;
  wordsContainer.scrollTop = scrollOffset > 0 ? scrollOffset : 0;
}

function calculateMetrics() {
  const now = Date.now();
  const elapsedSec = Math.max(1, (now - state.startTime) / 1000);
  const elapsedMinutes = elapsedSec / 60;

  const rawWpm = Math.round((state.totalTypedChars / 5) / elapsedMinutes) || 0;
  const netWpm = Math.max(0, Math.round((state.correctChars / 5) / elapsedMinutes)) || 0;
  const accuracy = state.totalTypedChars > 0
    ? Math.round((state.correctChars / state.totalTypedChars) * 100)
    : 100;

  return { netWpm, rawWpm, accuracy, elapsedSec };
}

function updateLiveHUD() {
  if (!state.started || state.finished) return;
  const metrics = calculateMetrics();

  $("liveWpm").textContent = String(metrics.netWpm).padStart(3, "0");
  $("liveRaw").textContent = String(metrics.rawWpm).padStart(3, "0");
  $("liveAccuracy").textContent = `${metrics.accuracy}%`;

  if (state.mode === "time") {
    const remaining = Math.max(0, state.timeLimit - Math.floor(metrics.elapsedSec));
    $("liveCounter").textContent = remaining;
    progressBar.style.width = `${((state.timeLimit - remaining) / state.timeLimit) * 100}%`;
    if (remaining <= 0) finishTest();
  } else {
    const wordsDone = state.currentWordIdx;
    const totalWords = state.words.length;
    $("liveCounter").textContent = `${wordsDone}/${totalWords}`;
    progressBar.style.width = `${(wordsDone / totalWords) * 100}%`;
    if (wordsDone >= totalWords) finishTest();
  }
}

function onTick() {
  if (!state.started || state.finished) return;

  const metrics = calculateMetrics();
  updateLiveHUD();

  const second = Math.floor(metrics.elapsedSec);
  if (!state.historySeconds.includes(second)) {
    state.historySeconds.push(second);
    state.historyWpm.push(metrics.netWpm);
    state.historyRaw.push(metrics.rawWpm);
    state.historyErrors.push(state.incorrectChars - state.lastSecondErrors);
    state.lastSecondErrors = state.incorrectChars;
  }
}

function startTest() {
  if (state.started) return;
  state.started = true;
  state.finished = false;
  state.startTime = Date.now();

  caretEl.classList.remove("blink");
  state.timerInterval = setInterval(onTick, 100);
}

function finishTest() {
  if (state.finished) return;
  state.finished = true;
  clearInterval(state.timerInterval);
  state.timerInterval = null;

  hiddenInput.blur();
  caretEl.classList.add("blink");
  Sfx.playFinishChime();

  const metrics = calculateMetrics();

  let consistency = 100;
  if (state.historyWpm.length > 2) {
    const avg = state.historyWpm.reduce((a, b) => a + b, 0) / state.historyWpm.length;
    const diffSquares = state.historyWpm.map(v => Math.pow(v - avg, 2));
    const variance = diffSquares.reduce((a, b) => a + b, 0) / state.historyWpm.length;
    consistency = Math.max(0, Math.round(100 - (Math.sqrt(variance) / (avg || 1)) * 100));
  }

  $("finalWpm").textContent = metrics.netWpm;
  $("finalAccuracy").textContent = `${metrics.accuracy}%`;
  $("finalRaw").textContent = metrics.rawWpm;
  $("finalConsistency").textContent = `${consistency}%`;
  $("finalChars").textContent = `${state.correctChars}/${state.incorrectChars}/${state.extraChars}`;
  $("finalTime").textContent = `${Math.round(metrics.elapsedSec)}s`;

  $("resultSummaryMode").textContent = `Mode: ${state.mode.toUpperCase()} • ${state.mode === 'time' ? state.timeLimit + 's' : state.words.length + ' words'}`;

  checkAndUpdatePb(metrics.netWpm);
  drawSpeedChart();

  resultsScreen.style.display = "block";
  resultsScreen.scrollIntoView({ behavior: "smooth", block: "nearest" });
}

hiddenInput.addEventListener("keydown", (e) => {
  if (state.finished) return;
  Sfx.init();

  if (e.getModifierState && e.getModifierState("CapsLock")) {
    capsBanner.classList.add("visible");
  } else {
    capsBanner.classList.remove("visible");
  }

  const currentWordEl = wordsContainer.children[state.currentWordIdx];
  if (!currentWordEl) return;

  if (e.code === "Space") {
    e.preventDefault();
    if (state.currentLetterIdx === 0) return;

    if (!state.started) startTest();
    Sfx.playSpaceClick();

    const letters = currentWordEl.querySelectorAll(".letter:not(.extra)");
    let wordHadError = false;
    for (let i = 0; i < letters.length; i++) {
      if (!letters[i].classList.contains("correct")) {
        wordHadError = true;
        break;
      }
    }
    if (wordHadError || currentWordEl.querySelectorAll(".extra").length > 0) {
      currentWordEl.classList.add("error-word");
    }

    state.currentWordIdx++;
    state.currentLetterIdx = 0;

    if (state.mode === "time" && state.currentWordIdx > state.words.length - 15) {
      const moreWords = generateWords();
      state.words = state.words.concat(moreWords);
      moreWords.forEach(w => {
        const wDiv = document.createElement("div");
        wDiv.className = "word";
        for (let ch of w) {
          const sp = document.createElement("span");
          sp.className = "letter";
          sp.textContent = ch;
          wDiv.appendChild(sp);
        }
        wordsContainer.appendChild(wDiv);
      });
    }

    updateCaretPosition();
    updateLiveHUD();
    return;
  }

  if (e.code === "Backspace") {
    e.preventDefault();
    if (!state.started) return;

    if (e.ctrlKey || e.altKey) {
      const extras = currentWordEl.querySelectorAll(".extra");
      extras.forEach(el => el.remove());

      const letters = currentWordEl.querySelectorAll(".letter");
      letters.forEach(l => l.className = "letter");
      state.currentLetterIdx = 0;
      Sfx.playKeyClick();
      updateCaretPosition();
      return;
    }

    if (state.currentLetterIdx > 0) {
      const letters = currentWordEl.querySelectorAll(".letter");
      const letterToUndo = letters[state.currentLetterIdx - 1];

      if (letterToUndo.classList.contains("extra")) {
        letterToUndo.remove();
        state.extraChars = Math.max(0, state.extraChars - 1);
      } else {
        letterToUndo.className = "letter";
      }

      state.currentLetterIdx--;
      Sfx.playKeyClick();
      updateCaretPosition();
    } else if (state.currentLetterIdx === 0 && state.currentWordIdx > 0) {
      const prevWordEl = wordsContainer.children[state.currentWordIdx - 1];
      if (prevWordEl && prevWordEl.classList.contains("error-word")) {
        state.currentWordIdx--;
        const letters = prevWordEl.querySelectorAll(".letter");
        state.currentLetterIdx = letters.length;
        prevWordEl.classList.remove("error-word");
        Sfx.playKeyClick();
        updateCaretPosition();
      }
    }
    return;
  }

  if (e.key.length === 1 && !e.ctrlKey && !e.metaKey) {
    e.preventDefault();
    if (!state.started) startTest();

    state.totalTypedChars++;
    const typedChar = e.key;
    const targetWord = state.words[state.currentWordIdx];
    const letters = currentWordEl.querySelectorAll(".letter:not(.extra)");

    if (state.currentLetterIdx < letters.length) {
      const targetChar = targetWord[state.currentLetterIdx];
      const letterSpan = letters[state.currentLetterIdx];

      if (typedChar === targetChar) {
        letterSpan.className = "letter correct";
        state.correctChars++;
        Sfx.playKeyClick();
      } else {
        letterSpan.className = "letter incorrect";
        state.incorrectChars++;
        Sfx.playError();
      }
      state.currentLetterIdx++;
    } else {
      if (currentWordEl.querySelectorAll(".extra").length < 10) {
        const extraSpan = document.createElement("span");
        extraSpan.className = "letter extra";
        extraSpan.textContent = typedChar;
        currentWordEl.appendChild(extraSpan);
        state.extraChars++;
        state.incorrectChars++;
        state.currentLetterIdx++;
        Sfx.playError();
      }
    }

    updateCaretPosition();
    updateLiveHUD();
  }

function drawSpeedChart() {
  const canvas = $("speedChart");
  const ctx = canvas.getContext("2d");
  const width = canvas.width;
  const height = canvas.height;

  ctx.clearRect(0, 0, width, height);

  const data = state.historyWpm;
  const rawData = state.historyRaw;
  const errors = state.historyErrors;

  if (data.length < 2) return;

  const maxVal = Math.max(80, ...rawData, ...data) + 10;
  $("chartStats").textContent = `PEAK: ${Math.max(...data)} WPM`;

  ctx.strokeStyle = "rgba(184, 242, 139, 0.15)";
  ctx.lineWidth = 1;
  ctx.setLineDash([4, 4]);

  for (let yVal = 20; yVal < maxVal; yVal += 20) {
    const y = height - (yVal / maxVal) * (height - 30) - 15;
    ctx.beginPath();
    ctx.moveTo(35, y);
    ctx.lineTo(width - 15, y);
    ctx.stroke();

    ctx.fillStyle = "rgba(184, 242, 139, 0.4)";
    ctx.font = "12px 'VT323', monospace";
    ctx.fillText(`${yVal}`, 8, y + 4);
  }
  ctx.setLineDash([]);

  const getX = (idx) => 40 + (idx / (data.length - 1)) * (width - 60);
  const getY = (val) => height - (val / maxVal) * (height - 30) - 15;

  ctx.strokeStyle = "rgba(184, 242, 139, 0.35)";
  ctx.lineWidth = 2;
  ctx.beginPath();
  rawData.forEach((val, i) => {
    const x = getX(i);
    const y = getY(val);
    if (i === 0) ctx.moveTo(x, y);
    else ctx.lineTo(x, y);
  });
  ctx.stroke();

  ctx.strokeStyle = getComputedStyle(document.documentElement).getPropertyValue('--primary').trim() || "#b8f28b";
  ctx.lineWidth = 3;
  ctx.shadowColor = ctx.strokeStyle;
  ctx.shadowBlur = 8;
  ctx.beginPath();
  data.forEach((val, i) => {
    const x = getX(i);
    const y = getY(val);
    if (i === 0) ctx.moveTo(x, y);
    else ctx.lineTo(x, y);
  });
  ctx.stroke();
  ctx.shadowBlur = 0;

  data.forEach((val, i) => {
    const x = getX(i);
    const y = getY(val);

    ctx.fillStyle = ctx.strokeStyle;
    ctx.beginPath();
    ctx.arc(x, y, 3, 0, Math.PI * 2);
    ctx.fill();

    if (errors[i] > 0) {
      ctx.fillStyle = "#ff5555";
      ctx.beginPath();
      ctx.arc(x, y - 8, 4, 0, Math.PI * 2);
      ctx.fill();
    }
  });
}
